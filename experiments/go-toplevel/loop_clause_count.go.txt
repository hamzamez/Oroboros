package main

// Fair three-clause comparison: the SAME algorithm, written idiomatically and
// written as loop/again. (j) earlier added a condition and so measured the
// condition, not the form.

// idiomatic Go, three outcomes.
func kHandThree(a []float64, k float64) int64 {
	for i := 0; i < len(a); i++ {
		if a[i] > k {
			return int64(i)
		}
		if a[i] < 0 {
			return -2
		}
	}
	return -1
}

// the same as loop/again would emit.
func lLoopThree(a []float64, k float64) int64 {
	var result int64
	i := int64(0)
	n := int64(len(a))
	for {
		if !(i < n) {
			result = -1
			break
		}
		if a[i] > k {
			result = i
			break
		}
		if a[i] < 0 {
			result = -2
			break
		}
		i = i + 1
	}
	return result
}
