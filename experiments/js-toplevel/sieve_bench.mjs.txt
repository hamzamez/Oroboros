import { genMain } from './jsieve.mjs';

// Idiomatic: start at i*i, step by i, skip with continue.
function handSieve(n) {
  const composite = new Array(n).fill(0);
  for (let i = 2; i * i < n; i++) {
    if (composite[i]) continue;
    for (let j = i * i; j < n; j += i) composite[j] = 1;
  }
  let count = 0;
  for (let k = 2; k < n; k++) if (!composite[k]) count++;
  return count;
}

// Hand-written, restricted to the loop shapes Oroboros can express.
function shapedSieve(n) {
  const composite = new Array(n).fill(0);
  for (let i = 0; i < n; i++) {
    if (i < 2) continue;
    if (composite[i]) continue;
    for (let j = 0; j < n; j++) if (j >= i * i && j % i === 0) composite[j] = 1;
  }
  let count = 0;
  for (let k = 0; k < n; k++) if (k >= 2 && !composite[k]) count++;
  return count;
}

function timeIt(name, reps, f) {
  f();
  let best = Infinity, got;
  for (let r = 0; r < 5; r++) {
    const t0 = process.hrtime.bigint();
    for (let k = 0; k < reps; k++) got = f();
    const d = Number(process.hrtime.bigint() - t0) / reps / 1000;
    if (d < best) best = d;
  }
  console.log(`${name.padEnd(14)} ${best.toFixed(1).padStart(12)} us   primes=${got}`);
  return best;
}

const h = timeIt('hand-written', 50, () => handSieve(20000));
const s = timeIt('same shape', 3, () => shapedSieve(20000));
const g = timeIt('GENERATED', 3, genMain);
console.log(`\nloop shape costs ${(s / h).toFixed(0)}x; emitter a further ${(g / s).toFixed(2)}x`);
console.log(`generated is ${(g / h).toFixed(0)}x slower overall`);
